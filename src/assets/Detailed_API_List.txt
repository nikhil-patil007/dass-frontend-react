  >>> Endpoint
GET / api / products /: slug

  >>> Response(Output)

  id: number
  name: string
  slug: string
  category: string
  price: number
  image: string
  description: string
  inStock: boolean
  rating: number
}

4 > Add to Cart
  >>> Endpoint
POST / api / cart

  >>> Body
{
  productId: number,
  qty ?: number // default 1
}

  >>> Response(Output)
{
  "success": true,
  "message": "Product added to cart successfully",
}

5 > Update Quantity
  >>> Endpoint
PATCH / api / cart /: itemId

  >>> Body
{
  qty: number
}

 >>> Response(Output)
{
  "success": true,
  "message": "Cart item quantity updated successfully",
}

11 > Login API(Core)

  >>> Endpoint
POST / api / auth / login

  >>> Body
type LoginRequest = {
  email: string
  password: string
  remember?: boolean
}

  >>> Response(Output)
type LoginResponse = {
  success: boolean
  data?: {
    user: {
      id: string
      name: string
      email: string
    }
    accessToken?: string   // if using JWT
  }
  error?: string
}

12 > Register API
  >>> Endpoint
POST / api / auth / signup

  >>> Body
type SignupRequest = {
  name: string
  email: string
  password: string
  confirm: string
}

  >>> Response
type SignupResponse = {
  success: boolean
  data?: {
    user: {
      id: string
      name: string
      email: string
    }
  }
  error?: string
}
